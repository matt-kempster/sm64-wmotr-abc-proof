  forall version le memory mario y trace final after out,
  le ! FGE._m = Some (Vptr mario Ptrofs.zero) ->
  Mem.load Mint32 memory mario 12 = Some (Vint ifg_twirl) ->
  Mem.load Mfloat32 memory mario 76 = Some (Vsingle y) ->
  Float32.cmp Clt y Float32.zero = true ->
  ocn_exec (Clight.globalenv (selected_clight_target version)) empty_env le memory
    (fn_body (ibag_determine_body version)) trace final after out ->
  trace = E0 /\ after = memory /\
    out = Out_return (Some (Vint Int.one, tuint)).
